## Why

- This is the THIRD correctness surface of Milestone M2 (the pure, zero-Pi, fully unit-tested core), sibling of `tokens.ts` (P1.M2.T1.S1, Complete) and `ledger.ts` (P1.M2.T2.S1, Complete). It is the **confabulation defense** (D/D17): the structured `mulligan:note` tells the resumed model the TRUE state of the abandoned span so it does not blindly redo side-effectful work (E5); the validator refuses a vacuous note; the two nudge texts ride in-tool_result and per-turn (zero extra requests — D4) to tell the agent rewind/shrink exist.
- spec/04 §3 + spec/05 §1 step 6: the rewind tool persists `note: NoteInput` into the `mulligan:rewind` marker AND calls `renderNote(...)` to produce the `mulligan:note` CustomMessage content (`pi.sendMessage({customType:"mulligan:note", content: <rendered>, display:true})`). validateNote (spec/05 §1 step 2) is the tool's step-2 gate BEFORE persisting — a vacuous note is refused (E9).
- spec/07 §1: the `tool_result` handler APPENDS `renderBloatReminder(...)` to a bloated result's content (advisory, D3 — appended not replaced, ~40 tokens once). spec/07 §2: the `context`/`injectNudge` path appends an EPHEMERAL `mulligan:nudge` custom message whose content is `renderDriftNudge(metric)` (NEVER persisted — assert zero `mulligan:nudge` entries on disk).
- Pure-helper design: notes.ts imports TYPE-ONLY from sibling pure modules (`./ledger.js` FileLedger, `./config.js` Granularity) — erased at compile, so it stays Pi-free and unit-testable. This is the pure-helper TIER (spec/11 §1), NOT a permanent zero-imports gate like tokens.ts/ledger.ts. See GOTCHA #2.

export function validateNote(note: NoteInput): NoteValidation {
  if (!isRecord(note)) return { valid: false, reason: NOTE_INVALID_REASON };
  for (const field of NOTE_FIELDS) {
    const value = readOwn(note, field);
    if (typeof value !== "string" || value.trim().length === 0) {
      return { valid: false, reason: NOTE_INVALID_REASON };
    }
  }
  return { valid: true };
}
export function renderNote(note: NoteInput, ledger: FileLedger, granularity: Granularity): string {
  const sections: string[] = [
    `## 🔄 Mulligan rewind (${granularity})`,
    `**What happened:** ${readNoteField(note, "what_happened")}`,
    `**Avoid:** ${readNoteField(note, "avoid")}`,
    `**Current true state:** ${readNoteField(note, "true_current_state")}`,
  ];
  for (const [tag, field] of LEDGER_BLOCKS) {
    const items = readLedgerList(ledger, field);
    if (items.length > 0) sections.push(`<${tag}>\n${items.join("\n")}\n</${tag}>`);
  }
  sections.push(`**Next:** ${readNoteField(note, "next")}`);
  return sections.join("\n\n");
}
export function renderBloatReminder(_toolName: string, bytes: number, thresholdBytes: number): string {
  const body = [
    `[mulligan] This result is ~${bytesToKb(bytes)} KB in your context (threshold ${bytesToKb(thresholdBytes)} KB).`,
    "If you don't need the full output going forward, call `mulligan_shrink` with a",
    'summary, or `mulligan_rewind(granularity:"last_tool_call_group")` if the whole',
    "call was a mistake. (The hidden/shrunk content stays on disk for the human.)",
  ].join("\n");
  return `\n---\n${body}`;
}
export function renderDriftNudge(metric: DriftNudgeInput): string {
  const delta = readDelta(metric);
  const hits = readBloatHits(metric);
  let firstLine: string;
  if (delta != null && hits.length > 0) {
    firstLine = `Previous turn added ~${kTokens(delta)} tokens to your context and produced ${hits.length} bloated ${resultWord(hits.length)}`;
  } else if (delta != null) {
    firstLine = `Previous turn added ~${kTokens(delta)} tokens to your context`;
  } else if (hits.length > 0) {
    firstLine = `Previous turn produced ${hits.length} bloated ${resultWord(hits.length)}`;
  } else {
    firstLine = "Previous turn changed your context"; 
  }
  return [
    `[mulligan] ${firstLine}.`,
    "If that growth was wasteful, consider `mulligan_rewind` (undo the turn) or `mulligan_shrink` (compact a result).",
    "Run `mulligan_audit` for a breakdown.",
  ].join("\n");
}
function readOwn(obj: unknown, key: string): unknown {
  if (!isRecord(obj)) return undefined;
  try { return obj[key]; } catch { return undefined; }
}
function bytesToKb(n: number): number {
  return Number.isFinite(n) && n >= 0 ? Math.round(n / 1024) : 0;
}
function kTokens(delta: number): string {
  return `${Math.round((delta / 1000) * 10) / 10}k`;
}
