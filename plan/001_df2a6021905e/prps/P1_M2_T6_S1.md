## Why

- This is the SIXTH and final correctness surface of Milestone M2 (Pure Core Library). It is the COMPOSITION layer: it takes the targeting resolvers (T5: resolveLastToolCallGroup / resolveLastTurn / resolveCheckpoint) and the pairing primitive (T4: partitionIntoUnits) and wires them into `filterPipeline` — the single pure function the Pi-coupled `context` handler (filter.ts, P1.M3.T2) calls as `return { messages: filterPipeline(event.messages, readMarkers(ctx), getConfig(), ctx.sessionManager.getBranch()) }`.
- **Architectural seam (architecture/external_deps.md §3.1):** `filterPipeline` handles **rewinds + shrinks + protectedOk ONLY**. Nudge injection (injectNudge) is applied in `filter.ts` glue (P1.M3.T2/T3) AFTER calling this pure pipeline, on the already-transformed array. This keeps `transforms.ts` Pi-FREE (it must NOT import `nudges.ts`, which has Pi-coupled handlers) and breaks the spec/11 build-order cycle (Step 5 filter → Step 7 nudges). The pipeline signature mirrors spec/06 §12 but WITHOUT the injectNudge line.
- Downstream consumers (do NOT build these here): filter.ts (P1.M3.T2) calls filterPipeline; markers.ts (P1.M3.T1) persists RewindMarker/ShrinkMarker whose fields filterPipeline + applyShrink read (the structural RewindMarkerLike/ShrinkMarkerLike types here are the read-slice contract); tools/shrink.ts (P1.M4.T2) ALSO imports `resolveShrinkTarget` directly for best-effort "matched now: yes/no" feedback at marker-creation time.
- Pairing correctness (api_verification.md §6.4): applyRewind only ever removes indices the resolvers computed from UNITS (T4) — so an assistant+results toolGroup is removed together; applyShrink preserves `toolCallId`/`role` via the spread — so a shrunk toolResult stays paired with its assistant call. The model API never sees an orphan.
- Purity: filterPipeline takes DATA only (messages, markers-as-data, config-as-data, branchEntries-as-data from getBranch()) — never `ctx`. Fully unit-testable without Pi/model/session.

## What

Five pure contract functions + five supporting types + three module-private helpers, appended to the zero-import `src/transforms.ts`. The pipeline composes transforms in a FIXED order (spec/03 §5; spec/06 §1/§12): (1) rewinds oldest-first, each resolving against the CURRENT (already-reduced) array then gap-closed via applyRewind, gated by protectedOk; (2) shrinks oldest-first via applyShrink on the post-rewind array; (3) return. NO nudge injection here.

export function applyRewind(messages, remove) {
  if (!Array.isArray(messages)) return [];
  if (!Array.isArray(remove) || remove.length === 0) return messages;       
  const removeSet = new Set();
  for (const r of remove) if (typeof r === "number" && !Number.isNaN(r)) removeSet.add(r);
  if (removeSet.size === 0) return messages;                                 
  return messages.filter((_msg, i) => !removeSet.has(i));                    
}
export function resolveShrinkTarget(messages, target) {
  if (!Array.isArray(messages)) return null;
  if (!isRecord(target)) return null;
  const callId = readOwn(target, "by_tool_call_id");
  if (typeof callId === "string" && callId.length > 0) {
    for (let i = 0; i < messages.length; i++) {
      const m = messages[i];
      if (!isRecord(m) || readOwn(m, "role") !== "toolResult") continue;
      if (readOwn(m, "toolCallId") === callId) return i;
    }
    return null;
  }
  const name = readOwn(target, "by_tool_name");
  if (typeof name === "string" && name.length > 0) {
    const wantFirst = readOwn(target, "occurrence") === "first";             
    let found = -1;
    for (let i = 0; i < messages.length; i++) {
      const m = messages[i];
      if (!isRecord(m) || readOwn(m, "role") !== "toolResult") continue;
      if (readOwn(m, "toolName") === name) { if (wantFirst) return i; found = i; }
    }
    return found === -1 ? null : found;
  }
  const needle = readOwn(target, "by_content_includes");
  if (typeof needle === "string") {
    for (let i = 0; i < messages.length; i++)
      if (stringifyContent(readOwn(messages[i], "content")).includes(needle)) return i;  
    return null;
  }
  return null;
}
export function applyShrink(messages, marker) {
  if (!Array.isArray(messages)) return [];
  if (!isRecord(marker)) return messages;
  const i = resolveShrinkTarget(messages, readOwn(marker, "target"));
  if (i === null || i < 0 || i >= messages.length) return messages;          
  const orig = messages[i];
  const text = typeof readOwn(marker, "replacement") === "string" ? readOwn(marker, "replacement") : "";
  const newContent = [{ type: "text", text }];
  const role = readOwn(orig, "role");
  let replacement;
  try { replacement = { ...orig, content: newContent }; }                    
  catch { replacement = { role: typeof role === "string" ? role : undefined, content: newContent }; }
  return messages.map((m, j) => (j === i ? replacement : m));                
}
export function protectedOk(messages, remove, config) {
  if (!Array.isArray(remove) || remove.length === 0) return true;
  if (!Array.isArray(messages)) return true;
  let protectFirstUser = true;
  const rewindCfg = isRecord(config) ? readOwn(config, "rewind") : undefined;
  const roles = isRecord(rewindCfg) ? readOwn(rewindCfg, "protectedRoles") : undefined;
  if (Array.isArray(roles) && roles.length > 0) protectFirstUser = roles.some((r) => r === "first:user");
  if (!protectFirstUser) return true;
  let iFirstUser = -1;
  for (let i = 0; i < messages.length; i++)
    if (isRecord(messages[i]) && readOwn(messages[i], "role") === "user") { iFirstUser = i; break; }
  if (iFirstUser === -1) return true;
  let minRemove = Infinity;
  for (const r of remove) if (typeof r === "number" && !Number.isNaN(r) && r < minRemove) minRemove = r;
  if (!Number.isFinite(minRemove)) return true;
  return minRemove > iFirstUser;
}
export function filterPipeline(messages, markers, config, branchEntries) {
  if (!Array.isArray(messages)) return [];
  const bundle = isRecord(markers) ? markers : undefined;
  const rewinds = Array.isArray(readOwn(bundle, "rewinds")) ? readOwn(bundle, "rewinds") : [];
  const shrinks = Array.isArray(readOwn(bundle, "shrinks")) ? readOwn(bundle, "shrinks") : [];
  let m = messages;
  for (const rw of stableSortBySeq(rewinds)) {
    const granularity = readOwn(rw, "granularity");
    const excludeId = typeof readOwn(rw, "excludeToolCallId") === "string" ? readOwn(rw, "excludeToolCallId") : undefined;
    let remove;
    if (granularity === "last_tool_call_group") {
      const units = partitionIntoUnits(m);                                  
      remove = resolveLastToolCallGroup(units, m, excludeId) ?? [];
    } else if (granularity === "last_turn") {
      remove = resolveLastTurn(m, readOwn(rw, "options"), excludeId).remove; 
    } else if (granularity === "checkpoint") {
      const cpName = typeof readOwn(rw, "checkpoint") === "string" ? readOwn(rw, "checkpoint") : "";
      remove = resolveCheckpoint(m, Array.isArray(branchEntries) ? branchEntries : [], cpName, excludeId)?.remove ?? [];
    } else {
      remove = [];
    }
    if (!protectedOk(m, remove, config)) continue;                          
    m = applyRewind(m, remove);
  }
  for (const sh of stableSortBySeq(shrinks)) m = applyShrink(m, sh);        
  return m;                                                                  
}
