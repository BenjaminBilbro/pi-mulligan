  if (!Array.isArray(units)) return null;
  const hasExclude = typeof excludeToolCallId === "string" && excludeToolCallId.length > 0;
  for (let k = units.length - 1; k >= 0; k--) {
    const unit = units[k];
    if (!isRecord(unit) || unit.kind !== "toolGroup" || !Array.isArray(unit.indices)) continue;
    if (hasExclude && assistantIssuedCall(messages, unit.indices, excludeToolCallId)) continue;
    return unit.indices; 
  }
  return null;
}
function assistantIssuedCall(messages, indices, callId) {
  if (!Array.isArray(messages)) return false;
  for (const i of indices) {
    const msg = messages[i];
    if (!isRecord(msg) || readOwn(msg, "role") !== "assistant") continue;
    const content = readOwn(msg, "content");
    if (!Array.isArray(content)) continue;
    for (const block of content) {
      if (!isRecord(block) || readOwn(block, "type") !== "toolCall") continue;
      if (readOwn(block, "id") === callId) return true;
    }
  }
  return false;
}
export function resolveLastTurn(messages, opts, excludeToolCallId?) {
  if (!Array.isArray(messages)) return { remove: [] };
  let iLastUser = -1;
  for (let i = 0; i < messages.length; i++)
    if (isRecord(messages[i]) && readOwn(messages[i], "role") === "user") iLastUser = i;
  if (iLastUser === -1) return { remove: [] };
  const nuclear = opts !== undefined && opts.to_previous_prompt === true; 
  if (nuclear) {
    let iFirstUser = -1;
    for (let i = 0; i < messages.length; i++)
      if (isRecord(messages[i]) && readOwn(messages[i], "role") === "user") { iFirstUser = i; break; }
    if (iFirstUser === iLastUser) return { remove: [] }; 
  }
  const rewindOwnIndices = new Set();
  const hasExclude = typeof excludeToolCallId === "string" && excludeToolCallId.length > 0;
  if (hasExclude) {
    const units = partitionIntoUnits(messages);
    for (const unit of units)
      if (unit.kind === "toolGroup" && assistantIssuedCall(messages, unit.indices, excludeToolCallId))
        for (const idx of unit.indices) rewindOwnIndices.add(idx); 
  }
  const remove = [];
  if (nuclear) remove.push(iLastUser);
  for (let j = iLastUser + 1; j < messages.length; j++) {
    if (rewindOwnIndices.has(j)) continue;
    if (isMulliganCustomMessage(messages[j])) continue; 
    remove.push(j);
  }
  return { remove };
}
function isMulliganCustomMessage(msg) {
  if (!isRecord(msg)) return false;
  const customType = readOwn(msg, "customType");
  return typeof customType === "string" && customType.startsWith("mulligan:");
}
export function resolveCheckpoint(messages, branchEntries, checkpointName, excludeToolCallId?) {
  if (!Array.isArray(messages) || !Array.isArray(branchEntries)) return null;
  if (typeof checkpointName !== "string" || checkpointName.length === 0) return null;
  const needle = `mulligan:checkpoint:${checkpointName}`;
  let targetId; 
  for (let i = branchEntries.length - 1; i >= 0; i--) {
    const e = branchEntries[i];
    if (!isRecord(e) || readOwn(e, "type") !== "label" || readOwn(e, "label") !== needle) continue;
    const tid = readOwn(e, "targetId");
    if (typeof tid === "string" && tid.length > 0) { targetId = tid; break; }
  }
  if (targetId === undefined) return null; 
  const ctxEntries = branchEntries.filter((e) =>
    isContextProducingType(isRecord(e) ? readOwn(e, "type") : undefined)); 
  let msgCursor = 0, iTarget = -1, found = false;
  for (const e of ctxEntries) {
    const y = entryMessageYield(e);
    if (y < 0) return null; 
    if (msgCursor + y > messages.length) return null; 
    if (isRecord(e) && readOwn(e, "id") === targetId) { iTarget = msgCursor + y - 1; found = true; break; }
    msgCursor += y;
  }
  if (!found) return null;
  const units = partitionIntoUnits(messages); 
  for (const unit of units) if (unit.indices.includes(iTarget)) { iTarget = Math.max(...unit.indices); break; }
  const rewindOwnIndices = new Set();
  const hasExclude = typeof excludeToolCallId === "string" && excludeToolCallId.length > 0;
  if (hasExclude)
    for (const unit of units)
      if (unit.kind === "toolGroup" && assistantIssuedCall(messages, unit.indices, excludeToolCallId))
        for (const idx of unit.indices) rewindOwnIndices.add(idx);
  const remove = [];
  for (let j = iTarget + 1; j < messages.length; j++) {
    if (rewindOwnIndices.has(j)) continue;
    if (isMulliganCustomMessage(messages[j])) continue;
    remove.push(j);
  }
  return { remove };
}
function entryMessageYield(entry) {
  const type = isRecord(entry) ? readOwn(entry, "type") : undefined;
  if (type === "message" || type === "custom_message" || type === "branch_summary") return 1;
  return -1; 
}
function isContextProducingType(type) {
  return type === "message" || type === "custom_message" || type === "branch_summary" || type === "compaction";
}
